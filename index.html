let lastResult = null; function getD(pickup, daughter) { const constants = { "4": {"1":10008, "2":10009, "3":10008}, "3": {"1":10000, "2":10000, "3":10000} }; return constants[pickup][daughter]; } function readNumber(id) { const value = document.getElementById(id) .value.trim().replace(",", "."); if (value === "") return null; const number = Number(value); return Number.isFinite(number) ? number : null; } function formatNumber(number) { return new Intl.NumberFormat("id-ID", { maximumFractionDigits: 2 }).format(number); } function calculate() { const charge = document.getElementById("charge").value.trim(); const pickup = document.getElementById("pickup").value; if (charge === "") { alert("Nomor Charge wajib diisi."); return; } const answers = []; for (let i = 1; i <= 3; i++) { const dp = readNumber("dp" + i); const dc = readNumber("dc" + i); if (dp === null || dc === null) { alert("DP dan DC Daughter " + i + " wajib diisi dengan angka."); return; } if (dp < 0 || dc < 0) { alert("DP dan DC Daughter " + i + " tidak boleh negatif."); return; } const D = getD(pickup, String(i)); const answer = D - (dp + dc); if (answer < 0) { alert("Hasil Daughter " + i + " negatif. Periksa DP dan DC."); return; } answers.push({dp, dc, answer}); } lastResult = {charge, pickup, answers}; for (let i = 1; i <= 3; i++) { document.getElementById("output" + i).textContent = formatNumber(answers[i - 1].answer) + " mm"; document.getElementById("result" + i).style.display = "block"; } document.getElementById("share").style.display = "block"; document.getElementById("shareTelegram").style.display = "block"; } function createShareMessage() { if (!lastResult) return ""; const r = lastResult; let message = "HOT SLAB LENGTH RESULT\n\n" + "Charge: " + r.charge + "\n" + "Pickup: " + r.pickup + "\n\n"; for (let i = 1; i <= 3; i++) { const d = r.answers[i - 1]; message += "DAUGHTER " + i + "\n" + "DP: " + formatNumber(d.dp) + " mm\n" + "DC: " + formatNumber(d.dc) + " mm\n" + "Hot Slab Length: " + formatNumber(d.answer) + " mm\n\n"; } return message; } function shareWhatsApp() { if (!lastResult) { alert("Silakan hitung terlebih dahulu."); return; } const message = createShareMessage(); window.open( "https://wa.me/?text=" + encodeURIComponent(message), "_blank" ); } function shareTelegram() { if (!lastResult) { alert("Silakan hitung terlebih dahulu."); return; } const message = createShareMessage(); const url = "https://t.me/share/url?url=" + "&text=" + encodeURIComponent(message); window.open(url, "_blank"); } function resetForm() { document.getElementById("charge").value = ""; document.getElementById("pickup").value = "4"; for (let i = 1; i <= 3; i++) { document.getElementById("dp" + i).value = ""; document.getElementById("dc" + i).value = ""; document.getElementById("result" + i).style.display = "none"; } document.getElementById("share").style.display = "none"; document.getElementById("shareTelegram").style.display = "none"; lastResult = null; }
